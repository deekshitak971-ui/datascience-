# Step 1: Import Required Libraries
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.datasets import fetch_california_housing
# Step 2: Load Dataset
housing = fetch_california_housing(as_frame=True)
df = housing.frame
print("✅Dataset Loaded Successfully!")
print(df.head())
print("\nDataset Shape:", df.shape)
print("\nData Types:\n", df.dtypes)
# Step 3: Create reports/figures Directory
os.makedirs("reports/figures", exist_ok=True)
print("\n📂 Directory 'reports/figures' created (if not already present).")
# Step 4: Generate Histograms for Each Numerical Feature
for col in df.select_dtypes(include=np.number).columns:
plt.figure(figsize=(8, 5))
sns.histplot(df[col], bins=30, kde=True, color='steelblue')
plt.title(f"Histogram of {col}", fontsize=14)
plt.xlabel(col)
plt.ylabel("Frequency")
plt.tight_layout()
filename = f"reports/figures/hist_{col}.png"
plt.savefig(filename)
plt.close()
print(f" Saved Histogram: {filename}")
# Step 5: Generate Box Plots for Outlier Detection
for col in df.select_dtypes(include=np.number).columns:
plt.figure(figsize=(8, 4))
sns.boxplot(x=df[col], color='lightcoral')
plt.title(f"Box Plot of {col}", fontsize=14)
plt.xlabel(col)
plt.tight_layout()
filename = f"reports/figures/box_{col}.png"
plt.savefig(filename)
plt.close()
print(f"📦 Saved Boxplot: {filename}")
